```python
def build_default_tool_runtime() -> ToolRuntime:
    """构造默认工具运行时（不含 hooks，由 AppRuntime 安装）。"""
    registry = ToolRegistry()
    registry.register(ReadTool())
    registry.register(WriteTool())
    registry.register(EditTool())
    registry.register(BashTool())
    registry.register(LsTool())
    registry.register(GrepTool())
    registry.register(FindTool())
    registry.register(WebSearchTool())
    registry.register(WebFetchTool())
    return ToolRuntime(registry)


def list_tools(runtime: ToolRuntime) -> str:
    """列出已注册工具。"""
    defs = runtime.get_all_definitions()
    if not defs:
        return "(无已注册工具)"
    lines = ["已注册工具:"]
    for d in defs:
        fn = d["function"]
        lines.append(f"  - {fn['name']}: {fn['description']}")
    return "\n".join(lines)


def build_long_term_memory(long_term_path: Path) -> TextLongTermMemory:
    """构造长期记忆存储。"""
    return TextLongTermMemory(long_term_path)
```

```python
import logging

log = logging.getLogger(__name__)


def build_default_mcp_manager(mcp_config: dict) -> tuple[object, int, int]:
    """
    构造并启动 MCP manager，返回 (manager, loaded_count, failed_count)。
    
    Args:
        mcp_config: MCP server 配置字典 {name: config}
    
    Returns:
        (McpServerManager, loaded_count, failed_count)
    """
    from mcp_integration import McpServerManager
    
    manager = McpServerManager()
    loaded = 0
    failed = 0
    
    for name, config in mcp_config.items():
        if getattr(config, 'disabled', False):
            log.debug("MCP server '%s' is disabled, skipping", name)
            continue
        
        try:
            tools = manager.start_server_sync(name, config)
            loaded += 1
            log.debug("MCP server '%s' loaded (%d tools)", name, len(tools))
        except Exception:
            failed += 1
            log.exception("MCP server '%s' failed", name)
    
    return manager, loaded, failed
```

```markdown
`app_runtime/` 负责创建、持有和刷新运行期资源：

  - `list_tools()` - 列出工具
  - `build_branch_summary()` - 生成分支摘要

  - `build_default_tool_runtime()` - 构建默认工具集
  - `build_default_mcp_manager()` - 构建 MCP 管理器
- **HookManager** — 工具 hooks 的安装与桥接（freshness guard）
- **AppStateStore** — 项目级运行期状态（插件启用状态）
- **SessionRuntime** — 会话上下文准备、自动压缩、agent 对话重载
- **SkillManager** — 组合 PluginManager + SkillRegistry
- **EventBus** — 运行期生命周期事件发布

调用方通过 `app_runtime.X` 访问能力，不再通过门面转发。
```
